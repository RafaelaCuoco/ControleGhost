using System.Text.Json;
using System.Xml;
using System.Text;

public static class PoliteConverter
{
    public static string ConvertJsonToXml(string json)
    {
        if (string.IsNullOrWhiteSpace(json)) return "<Root/>";

        var sb = new StringBuilder();
        // Configuração leve do XmlWriter
        using var xmlWriter = XmlWriter.Create(sb, new XmlWriterSettings 
        { 
            OmitXmlDeclaration = true, 
            Indent = false // Sem indentação para economizar bytes de rede
        });

        // JsonDocument é "Zero-Allocation" (não gera lixo para o Garbage Collector)
        using var doc = JsonDocument.Parse(json);
        
        xmlWriter.WriteStartElement("InfNFSe"); // Tag raiz do XML
        
        // Função recursiva simples para mapear o JSON para XML
        WriteElement(xmlWriter, "Data", doc.RootElement); 
        
        xmlWriter.WriteEndElement();
        xmlWriter.Flush();
        
        return sb.ToString();
    }

    private static void WriteElement(XmlWriter writer, string name, JsonElement element)
    {
        switch (element.ValueKind)
        {
            case JsonValueKind.Object:
                writer.WriteStartElement(name);
                foreach (var prop in element.EnumerateObject())
                    WriteElement(writer, prop.Name, prop.Value);
                writer.WriteEndElement();
                break;
            case JsonValueKind.Array:
                writer.WriteStartElement(name);
                foreach (var item in element.EnumerateArray())
                    WriteElement(writer, "Item", item); // Ou o nome específico do array
                writer.WriteEndElement();
                break;
            default:
                writer.WriteElementString(name, element.ToString());
                break;
        }
    }
}
