using Microsoft.Data.SqlClient;
using System.Data;
using System.Text;
using System.Text.Json;
using System.Xml;

public class GhostModeEtlPipeline
{
    private readonly string _connectionString;
    private readonly BandwidthThrottler _throttler;
    
    // Configurações de "Educação" Extrema
    private const int BATCH_SIZE = 500;         // Lotes menores
    private const int MAX_CONCURRENT_WRITES = 1; // Apenas 1 escrita no banco por vez
    private const int MAX_MB_PER_SECOND = 2;    // Limite de rede: 2 MB/s (Bem baixo!)

    public GhostModeEtlPipeline(string connectionString)
    {
        _connectionString = connectionString;
        _throttler = new BandwidthThrottler(MAX_MB_PER_SECOND);
    }

    // Método público que garante a execução em Thread de Baixa Prioridade
    public async Task ExecutePoliteAsync(CancellationToken cancellationToken)
    {
        // TaskCreationOptions.LongRunning cria uma thread dedicada, não usando a ThreadPool
        await Task.Factory.StartNew(async () =>
        {
            // 1. LIMITAR PRIORIDADE DA THREAD
            Thread.CurrentThread.Priority = ThreadPriority.Lowest;
            Thread.CurrentThread.IsBackground = true;
            
            await RunPipelineCoreAsync(cancellationToken);
            
        }, cancellationToken, TaskCreationOptions.LongRunning, TaskScheduler.Default).Unwrap();
    }

    private async Task RunPipelineCoreAsync(CancellationToken ct)
    {
        // 2. SEMAPHORESLIM PARA BACKPRESSURE (Controle de Concorrência)
        using var semaphore = new SemaphoreSlim(MAX_CONCURRENT_WRITES, MAX_CONCURRENT_WRITES);

        using var connection = new SqlConnection(_connectionString);
        await connection.OpenAsync(ct);

        string query = "SELECT ServicoTomadoId, TagInfNFSe FROM NFe.ServicoTomadoAmbienteNacionalJson WITH (READPAST) ORDER BY ServicoTomadoId";
        
        using var cmd = new SqlCommand(query, connection);
        // SequentialAccess é obrigatório para não carregar o blob na memória
        using var reader = await cmd.ExecuteReaderAsync(CommandBehavior.SequentialAccess, ct);

        var batchTable = CreateBatchTable();
        int rowCount = 0;
        var tasks = new List<Task>();

        while (await reader.ReadAsync(ct))
        {
            int id = reader.GetInt32(0);
            string json = reader.GetString(1); 
            
            int jsonBytes = Encoding.UTF8.GetByteCount(json);
            
            // 3. THROTTLING DE LEITURA (Limita a banda de entrada)
            await _throttler.ThrottleAsync(jsonBytes, ct);

            // Converte JSON para XML (Zero-allocation)
            string xml = ConvertJsonToXml(json);
            int xmlBytes = Encoding.UTF8.GetByteCount(xml);

            batchTable.Rows.Add(id, xml);
            rowCount++;

            // 4. THROTTLING DE ESCRITA (Limita a banda de saída)
            await _throttler.ThrottleAsync(xmlBytes, ct);

            if (rowCount >= BATCH_SIZE)
            {
                // Espera o semáforo liberar (garante que não sobrecarrega o DB com writes paralelos)
                await semaphore.WaitAsync(ct); 
                
                var tableToWrite = batchTable;
                batchTable = CreateBatchTable(); // Cria nova tabela para o próximo lote
                rowCount = 0;

                // Dispara a escrita em background, mas limitada pelo semáforo
                var writeTask = Task.Run(async () =>
                {
                    try { await WriteBatchAsync(tableToWrite, ct); }
                    finally { semaphore.Release(); } // Libera o semáforo
                }, ct);
                
                tasks.Add(writeTask);
            }
        }

        // Processa o último lote
        if (rowCount > 0)
        {
            await semaphore.WaitAsync(ct);
            var writeTask = Task.Run(async () =>
            {
                try { await WriteBatchAsync(batchTable, ct); }
                finally { semaphore.Release(); }
            }, ct);
            tasks.Add(writeTask);
        }

        // Aguarda todas as escritas terminarem
        await Task.WhenAll(tasks);
    }

    private async Task WriteBatchAsync(DataTable table, CancellationToken ct)
    {
        using var bulkCopy = new SqlBulkCopy(_connectionString)
        {
            DestinationTableName = "NFe.ServicoTomadoAmbienteNacionalXml",
            BatchSize = BATCH_SIZE,
            BulkCopyTimeout = 600 // Timeout alto pois estamos indo devagar
        };
        
        bulkCopy.ColumnMappings.Add("ServicoTomadoId", "ServicoTomadoId");
        bulkCopy.ColumnMappings.Add("TagInfNFSeXml", "TagInfNFSeXml");

        await bulkCopy.WriteToServerAsync(table, ct);
    }

    private DataTable CreateBatchTable()
    {
        var table = new DataTable();
        table.Columns.Add("ServicoTomadoId", typeof(int));
        table.Columns.Add("TagInfNFSeXml", typeof(string));
        return table;
    }

    private string ConvertJsonToXml(string json) { /* ... */ return "<xml/>"; }
}
